\begin{frame}
  \begin{definition}[Homomorphism (同态)]
    Let $(G, \cdot)$ and $(G', \ast)$ be two groups.
    Let $\phi$ be a function such that
    \[
      \forall a, b \in G\; \phi(a \cdot b) = \phi(a) \ast \phi(b).
    \]
    Then $\phi$ is a \red{homomorphism} from $G$ to $G'$.
  \end{definition}

  \pause
  \vspace{0.50cm}
  \begin{center}
    If $\phi$ is a bijection, then $G$ and $G'$ are called \red{isomorphic}.
    \[
      \phi: G \cong G'
    \]
  \end{center}
\end{frame}

\begin{frame}{}
  \begin{theorem}
    Suppose that \blue{$\phi$ is a homomorphism from $G$ to $G'$}. \\
    Let $e$ and $e'$ are identities of $G$ and $G'$, respectively.

    \begin{enumerate}[(1)]
      \setlength{\itemsep}{6pt}
      \item $\phi(e) = e'$
      \item \teal{$\phi(a^{-1}) = (\phi(a))^{-1}$}
    \end{enumerate}
  \end{theorem}

  \pause
  \[
    e' \phi(e) = \phi(e) = \purple{\phi(e e)} = \phi(e) \phi(e) \implies \phi(e) = e'
  \]

  \pause
  \[
    \phi(a)\phi(a^{-1}) = \purple{\phi(aa^{-1})} = \phi(e) = e' = \phi(a)(\phi(a))^{-1}
  \]
\end{frame}

\begin{frame}
  \begin{theorem}
    Suppose that \blue{$\phi$ is a homomorphism from $G$ to $G'$}. \\
    \begin{enumerate}[(1)]
      \item
        \[
          H \le G \implies \phi(H) \le \phi(G)
        \]
      \item
        \[
          \purple{H \triangleleft G \implies \phi(H) \triangleleft \phi(G)}
        \]
      \pause
      \item
        \[
          K \le G' \implies \phi^{-1}(K) \le G
        \]
      \item
        \[
          K \triangleleft G' \implies \phi^{-1}(K) \triangleleft G
        \]
    \end{enumerate}
  \end{theorem}
\end{frame}

\begin{frame}
  \[
    \purple{H \triangleleft G \implies \phi(H) \triangleleft \phi(G)}
  \]

  \pause
  \[
    \forall a \in G, h \in H\; aha^{-1} \in H
  \]
  \pause
  \[
    \phi(a) \in \phi(G), \quad \phi(h) \in \phi(H)
  \]

  \pause
  \[
    \teal{\phi(a) \phi(h) \phi(a)^{-1}}
      = \phi(a) \phi(h) \phi(a^{-1})
      = \phi(aha^{-1}) \teal{\;\in \phi(H)}
  \]
\end{frame}

\begin{frame}
  \[
    \purple{K \triangleleft G'} \implies \phi^{-1}(K) \blue{\;\triangleleft\; G}
  \]

  \pause
  \begin{definition}[同态核 (Kernel)]
    Suppose that \blue{$\phi$ is a homomorphism from $G$ to $G'$}. \\[3pt]
    Let $e'$ be the identity of $G'$. Then \purple{$\set{e'} \triangleleft G'$}.
    \[
      \phi^{-1}(\set{e'}) = \set{a \in G \mid \phi(a) = e'} \blue{\;\triangleleft\; G}
    \]
    is the \red{kernel} of $\phi$, denoted $\text{Ker}\; \phi$.
  \end{definition}
\end{frame}

\begin{frame}
  \begin{theorem}[Fundamental Homomorphism Theorem (群同态基本定理)]
    Suppose that \blue{$\phi$ is a homomorphism from $G$ to $G'$}. Then
    \[
      G/\text{Ker}\; \phi \cong \phi(G).
    \]
  \end{theorem}

  \pause
  \begin{center}
    \red{同态核可以看作群 $G$ 与其同态像 $\phi(G)$ 之间{\bf 相似程度}的一种度量}
  \end{center}
